cmake_minimum_required(VERSION 3.16)
project(rviz_snapshot LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_executable(rviz_snapshot snapshot.cpp)
target_compile_features(rviz_snapshot PRIVATE cxx_std_17)
target_link_libraries(rviz_snapshot Qt5::Widgets)
ament_target_dependencies(rviz_snapshot rclcpp rviz_common rviz_rendering)
