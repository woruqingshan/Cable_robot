cmake_minimum_required(VERSION 3.16)
project(cable_collision_backend VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
find_package(resource_retriever REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(collision_engine src/engine.cpp)
target_include_directories(collision_engine PUBLIC include)
ament_target_dependencies(collision_engine moveit_core moveit_ros_planning urdf srdfdom resource_retriever)
target_link_libraries(collision_engine nlohmann_json::nlohmann_json OpenSSL::Crypto)
target_compile_options(collision_engine PRIVATE -Wall -Wextra -Wpedantic)
add_executable(collision_backend src/main.cpp)
target_link_libraries(collision_backend collision_engine)
enable_testing()
add_executable(native_tests tests/native_tests.cpp)
target_link_libraries(native_tests collision_engine)
add_test(NAME native_collision COMMAND native_tests)
